def stable_unique(items):
    seen = []
    result = []
    for item in items:
        if item not in seen:

  • Keep every element exactly once in order.

Reference examples — trace only

  {
    "input": [
      [
        1,
        2,
        3,
        4,
        5
      ],
      2
    ],
    "output": [
      [
        1,
        2
      ],
      [
        3,

Exact response

def chunks(items, size):  
    if size <= 0:  
        raise ValueError  
    result = []  
    for i in range(0, len(items), size):  
        result.append(items[i:i+size])  
    return result
